  <div class="container">
    <h1>Feature Prioritization Assistant</h1>
    <p class="subtitle">Score your features, then let Claude rank your backlog with strategic rationale.</p>

    <div class="section-label">Product goal or context (optional)</div>
    <input type="text" class="goal-input" id="goal-input" placeholder="e.g. Increase self-service adoption for small business customers" />

    <div class="header-row">
      <span class="col-header">Feature</span>
      <span class="col-header">Impact (1–5)</span>
      <span class="col-header">Effort (1–5)</span>
      <span class="col-header">Confidence (1–5)</span>
      <span></span>
    </div>

    <div id="features-list"></div>
    <button class="add-btn" onclick="addFeature()">+ Add feature</button>
    <button class="run-btn" id="run-btn" onclick="runPrioritization()">Prioritize my backlog</button>

    <hr class="divider" id="results-divider" style="display:none">
    <div id="results-area"></div>
  </div>

  <script>
    let featureCount = 0;

    function addFeature(name = "", impact = 3, effort = 3, confidence = 3) {
      featureCount++;
      const id = featureCount;
      const list = document.getElementById("features-list");
      const row = document.createElement("div");
      row.className = "feature-row";
      row.id = "feature-row-" + id;
      row.innerHTML = `
        <input type="text" placeholder="Feature name" value="${name}" id="feat-name-${id}" />
        <select id="feat-impact-${id}">${[1,2,3,4,5].map(v => `<option value="${v}" ${v==impact?"selected":""}>${v}</option>`).join("")}</select>
        <select id="feat-effort-${id}">${[1,2,3,4,5].map(v => `<option value="${v}" ${v==effort?"selected":""}>${v}</option>`).join("")}</select>
        <select id="feat-confidence-${id}">${[1,2,3,4,5].map(v => `<option value="${v}" ${v==confidence?"selected":""}>${v}</option>`).join("")}</select>
        <button class="remove-btn" onclick="removeFeature(${id})">×</button>
      `;
      list.appendChild(row);
    }

    function removeFeature(id) {
      const row = document.getElementById("feature-row-" + id);
      if (row) row.remove();
    }

    function getFeatures() {
      const rows = document.querySelectorAll("[id^='feature-row-']");
      const features = [];
      rows.forEach(row => {
        const id = row.id.replace("feature-row-", "");
        const name = document.getElementById("feat-name-" + id)?.value?.trim();
        if (!name) return;
        features.push({
          name,
          impact: parseInt(document.getElementById("feat-impact-" + id)?.value || 3),
          effort: parseInt(document.getElementById("feat-effort-" + id)?.value || 3),
          confidence: parseInt(document.getElementById("feat-confidence-" + id)?.value || 3),
        });
      });
      return features;
    }

    async function runPrioritization() {
      const features = getFeatures();
      if (features.length < 2) { alert("Please add at least 2 features to prioritize."); return; }
      const goal = document.getElementById("goal-input")?.value?.trim();
      const btn = document.getElementById("run-btn");
      const resultsArea = document.getElementById("results-area");
      const divider = document.getElementById("results-divider");
      btn.disabled = true;
      btn.textContent = "Analyzing...";
      divider.style.display = "block";
      resultsArea.innerHTML = '<p class="loading-text">Claude is ranking your features...</p>';

      const featureList = features.map((f, i) =>
        `${i+1}. "${f.name}" — Impact: ${f.impact}/5, Effort: ${f.effort}/5, Confidence: ${f.confidence}/5`
      ).join("\n");

      const prompt = `You are a senior product manager helping prioritize a feature backlog.
${goal ? `Product goal: ${goal}` : ""}

Here are the features with their scores (Impact = business/user value, Effort = implementation cost where lower is easier, Confidence = certainty that impact estimate is correct):

${featureList}

Rank these features from highest to lowest priority. For each, provide a 1-2 sentence rationale explaining the rank based on the scores and any strategic reasoning.

Respond ONLY with valid JSON. No markdown, no backticks, no explanation outside the JSON. Format:
{
  "rankedFeatures": [
    { "name": "feature name", "rank": 1, "impact": 0, "effort": 0, "confidence": 0, "rationale": "your explanation" }
  ],
  "summary": "1-2 sentence overall strategic summary"
}`;

      try {
        const response = await fetch("https://api.anthropic.com/v1/messages", {
          method: "POST",
          headers: { "Content-Type": "application/json" },
          body: JSON.stringify({ model: "claude-sonnet-4-20250514", max_tokens: 1000, messages: [{ role: "user", content: prompt }] })
        });
        const data = await response.json();
        const text = data.content?.map(b => b.text || "").join("").replace(/```json|```/g, "").trim();
        const parsed = JSON.parse(text);
        renderResults(parsed);
      } catch (err) {
        resultsArea.innerHTML = '<p class="error-text">Something went wrong. Please try again.</p>';
      } finally {
        btn.disabled = false;
        btn.textContent = "Re-prioritize";
      }
    }

    function scoreColor(type, val) {
      if (type === "impact") { if (val >= 4) return "background:#EAF3DE;color:#3B6D11"; if (val >= 3) return "background:#FAEEDA;color:#854F0B"; return "background:#FCEBEB;color:#A32D2D"; }
      if (type === "effort") { if (val <= 2) return "background:#EAF3DE;color:#3B6D11"; if (val <= 3) return "background:#FAEEDA;color:#854F0B"; return "background:#FCEBEB;color:#A32D2D"; }
      return "background:#E6F1FB;color:#185FA5";
    }

    function renderResults(data) {
      const resultsArea = document.getElementById("results-area");
      let html = "";
      if (data.summary) html += `<div class="summary-box">${data.summary}</div>`;
      data.rankedFeatures.forEach(f => {
        html += `<div class="result-card">
          <div class="result-rank">Priority #${f.rank}</div>
          <div class="result-name">${f.rank <= 3 ? ["🥇","🥈","🥉"][f.rank-1] + " " : ""}${f.name}</div>
          <div class="result-scores">
            <span class="score-pill" style="${scoreColor("impact", f.impact)}">Impact ${f.impact}/5</span>
            <span class="score-pill" style="${scoreColor("effort", f.effort)}">Effort ${f.effort}/5</span>
            <span class="score-pill" style="${scoreColor("confidence", f.confidence)}">Confidence ${f.confidence}/5</span>
          </div>
          <div class="result-rationale">${f.rationale}</div>
        </div>`;
      });
      resultsArea.innerHTML = html;
    }

    addFeature("User notifications", 4, 2, 4);
    addFeature("CSV export", 3, 2, 5);
    addFeature("Custom dashboards", 5, 5, 3);
    addFeature("SSO integration", 4, 4, 4);
  </script>
